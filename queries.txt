__________CAMPUS APP ACTIVITY BY STUDENT______________

CREATE KEYSPACE campus
WITH replication = {
    'class': 'NetworkTopologyStrategy',
    'replication_factor': 1
};


USE campus;


CREATE TABLE activity_by_student (
    student_id text,
    event_time timestamp,
    event_type text,
    course_code text,
    details text,
    PRIMARY KEY ((student_id), event_time)
)
WITH CLUSTERING ORDER BY (event_time DESC);
